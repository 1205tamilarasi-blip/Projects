    <script>
        function saveStudent() {

            // Get values from the form
            let name = document.getElementById("name").value;
            let age = document.getElementById("age").value;
            let course = document.getElementById("course").value;
            let email = document.getElementById("email").value;

            // Get selected gender
            let gender = document.querySelector(
                'input[name="gender"]:checked'
            );

            // Check whether all details are entered
            if (name === "" || age === "" || email === "" || !gender) {
                alert("Please fill all the details!");
                return;
            }

            // Create a new row
            let table = document.getElementById("tableBody");
            let newRow = table.insertRow();

            // Add student details to the row
            newRow.innerHTML = `
                <td>${name}</td>
                <td>${age}</td>
                <td>${course}</td>
                <td>${gender.value}</td>
                <td>${email}</td>
                <td>
                    <button class="delete-btn" onclick="deleteStudent(this)">
                        Delete
                    </button>
                </td>
            `;

            // Clear the form after saving
            document.getElementById("name").value = "";
            document.getElementById("age").value = "";
            document.getElementById("email").value = "";
            document.querySelectorAll('input[name="gender"]').forEach(
                radio => radio.checked = false
            );
        }

        // Function to delete a student row
        function deleteStudent(button) {
            button.parentElement.parentElement.remove();
        }
    </script>
